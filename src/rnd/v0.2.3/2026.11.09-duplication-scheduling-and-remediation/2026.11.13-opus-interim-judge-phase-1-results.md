## 3. Against the human labels

Same rules as the proof: recall over the 80 planted and monolith true pairs; precision over flagged pairs that carry a label.

| | Fable pass 0 | Opus pass 0 |
|---|---|---|
| Found (bar 0.80) | 72/80 = 0.900 | 71/80 = 0.8875 |
| Correct (bar 0.80) | 127/157 = 0.809 | 121/143 = 0.846 |
| Flags with no label yet | 0 | 5 |

- **Unlabelled Opus flags:** 5 flagged pairs have no label, so the 143 leaves them out. If all five were wrong, correct would be 121/148 = 0.818, still above 0.80.
- **Strata, found (Fable / Opus):** Fable-planted 45/45 and 45/45; monolith neighbour 4/5 and 3/5; monolith structural 10/15 and 10/15; **Opus work-alike 13/15 and 13/15**. The work-alike stratum shows no difference between the models, so there is no sign of Opus judging its own writing differently (15 pairs; a small sample).
- **McNemar on the 80 true pairs** (the fair test): Fable-only 2, Opus-only 1, exact p = 1.0.
- **McNemar on all 215 labelled pairs:** Fable-only 24, Opus-only 10, p = 0.024. **Do not read this as Fable being better.** The labelled pairs were picked from Fable's own flags, so pairs only Fable flagged are over-represented, and Opus's new flags are mostly unlabelled.

## Appendix: the map builder, as run

```python
"""Build { ledger key prefix | candidate : [ query id, candidate id ] } by replaying judge_all against the official ledger."""
import json, pathlib, sys
import cosa.utils.util as cu
from cosa.repo.semdup import config as cfg_mod, frozen, judge, pipeline
from cosa.repo.semdup.proof import proof_cli as pc, proof_run

RUN  = pathlib.Path( "/mnt/DATA01/include/www.deepily.ai/projects/lupin/io/semdup-labelling/2026.10.25-run5" )
LED  = pathlib.Path( "/mnt/DATA01/include/www.deepily.ai/projects/lupin/io/semdup/proof-official-2026.10.31/judge-ledger.jsonl" )
OUT  = pathlib.Path( sys.argv[1] )
vals = {}
for line in LED.read_text().splitlines():
    r = json.loads( line )
    if "key" in r: vals[ r[ "key" ] ] = r[ "value" ]

class Rec:
    def __init__( self ): self.hits = []; self.misses = []
    def get( self, key ):
        v = vals.get( key )
        ( self.hits if v is not None else self.misses ).append( key )
        return v
    def put( self, *a, **k ): raise RuntimeError( "would call the model" )

stash = {}
orig_pair = judge._pair
def pair( d, cands ):
    stash[ "id" ] = d[ "id" ]
    return orig_pair( d, cands )
judge._pair = pair
orig_key = judge.harness_runner.ledger_key
keymap = {}
def lkey( *a, **k ):
    key = orig_key( *a, **k )
    keymap[ key ] = stash[ "id" ]
    return key
judge.harness_runner.ledger_key = lkey

from cosa.repo.semdup import body_vectors
body_vectors.engine_tag = lambda c: "d093d02e5bb485ea"
class NoEmbed:
    def __getattr__( self, n ): raise RuntimeError( "the cache missed: would call the embedding service" )
root = cu.get_project_root()
cfg  = cfg_mod.verify_frozen( frozen.frozen_path(), frozen.live_config() )
print( "root", root, "model", cfg[ "model_id" ], file=sys.stderr )
parts = pc.live_parts( root, RUN, cfg, 4 )
spec  = pc.build_spec( parts[ "run" ], parts[ "mono_defs" ], parts[ "distractors" ], {}, [], {}, [] )
pool  = proof_run.build_pool( spec, spec[ "seed" ] )
res   = {}
for slot in ( 0, 1 ):
    rec = Rec()
    w   = { **parts[ "wiring" ], "ledger": rec, "client": NoEmbed() }
    out = pipeline.judge_all( pool[ "new" ], pool[ "defs" ], cfg, w, slot )
    print( "slot", slot, "hits", len( rec.hits ), "misses", len( rec.misses ), "unjudged", len( out[ "unjudged" ] ), file=sys.stderr )
    res[ slot ] = out

from cosa.repo.semdup.proof import label_load
P = RUN / "packets"
SUPP = RUN / "supplement-proof1"
supp = { "key": SUPP / "supplement-key.json", "labels": [ P / "labels-rick-supp.jsonl", SUPP / "packets" / "labels-reviewer-2-supp.jsonl" ], "third": P / "labels-jason-supp.jsonl" }
key = RUN / "key" / "answer-key.json"
labels = label_load.resolved_labels( P / "labels-rick.jsonl", P / "labels-reviewer-2.jsonl", key, None, P / "labels-jason.jsonl", supp )
prints = label_load.fingerprints( [ P / "labels-rick.jsonl", P / "labels-reviewer-2.jsonl" ], None, P / "labels-jason.jsonl", key, root, supp )
rec2 = json.load( open( "/mnt/DATA01/include/www.deepily.ai/projects/lupin/io/semdup/proof-official-2026.10.31/2.json" ) )
print( "answer_key fp match", prints[ "answer_key" ] == rec2[ "provenance" ][ "answer_key" ], "third", prints.get( "third_labels" ) == rec2[ "provenance" ][ "third_labels" ], "supp", prints[ "supplement" ][ "key" ] == rec2[ "provenance" ][ "supplement" ][ "key" ], file=sys.stderr )
canon = pool[ "canon" ]
failures = set( rec2[ "plant_failures" ][ "pairs" ] )
known = {}
for p_ in spec[ "planted" ]:
    k = proof_run.label_key( p_[ "rewrite" ][ "id" ], p_[ "original" ][ "id" ] )
    if k not in failures: known[ k ] = p_[ "stratum" ]
for p_ in spec[ "monolith_pairs" ]: known[ proof_run.label_key( *p_[ "ids" ] ) ] = "monolith:" + p_[ "source" ]
prefix_of = { qid: "|".join( k.split( "|" )[ :4 ] ) for k, qid in keymap.items() }
pm = {}
for v in res[ 0 ][ "verdicts" ]:
    pm[ prefix_of[ v[ "id" ] ] + "|" + v[ "candidate" ] ] = proof_run.label_key( canon[ v[ "id" ] ], canon[ v[ "candidate" ] ] )
flags0 = { proof_run.label_key( canon[ a ], canon[ b ] ) for a, b in proof_run.cluster.join_pairs( res[ 0 ][ "verdicts" ] ) } - failures
true = sorted( k for k in known if labels.get( k ) is True )
scored = flags0 & set( labels )
print( "VALIDATE true", len( true ), "found", len( set( true ) & flags0 ), "flags", len( scored ), "correct", sum( 1 for k in scored if labels[ k ] ), "(record: 80 / 72 / 157 / 127)", file=sys.stderr )
print( "label keys mapped", len( set( pm.values() ) & set( labels ) ), "of", len( labels ), file=sys.stderr )
OUT.write_text( json.dumps( { "pair_map": pm, "labels": labels, "strata": { k: known[ k ] for k in true } } ) )
print( "pairs mapped", len( pm ), file=sys.stderr )
raise SystemExit
```
