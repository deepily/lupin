**Storage Pattern** (following deep research pattern):

```python
import cosa.utils.util as cu
from cosa.config.configuration_manager import ConfigurationManager

class DataFrameStorage:
    def __init__( self, user_email: str, config_mgr: ConfigurationManager ):
        self.user_email = user_email
        output_path = config_mgr.get( "crud for dataframes output path" )
        self.user_dir = cu.get_project_root() + output_path + f"/{user_email}"
        os.makedirs( self.user_dir, exist_ok=True )

    def get_parquet_path( self, schema_type: str ) -> str:
        return f"{self.user_dir}/{schema_type}.parquet"

    def load_df( self, schema_type: str ) -> pd.DataFrame:
        path = self.get_parquet_path( schema_type )
        if os.path.exists( path ):
            return pd.read_parquet( path )
        return self._create_empty_df( schema_type )

    def save_df( self, df: pd.DataFrame, schema_type: str ) -> None:
        df.to_parquet( self.get_parquet_path( schema_type ), index=False )
```

**Agent Class** (following AgentBase pattern from `math_agent.py`):

```python
from cosa.agents.agent_base import AgentBase

class CrudForDataFramesAgent( AgentBase ):
    def __init__( self,
                  question: str = "",
                  user_id: str = "",
                  user_email: str = "",
                  session_id: str = "",
                  debug: bool = False ):

        super().__init__(
            routing_command="agent router go to crud for dataframes",
            question=question,
            user_id=user_id,
            user_email=user_email,
            session_id=session_id,
            debug=debug
        )

        # Initialize user-specific storage
        self.storage = DataFrameStorage( user_email, self.config_mgr )

        # Build prompt with user's list metadata
        self.prompt = self._build_prompt()
        self.xml_response_tag_names = [ "operation", "target_list", "schema_type",
                                        "confidence", "fields", "filters" ]

    def _build_prompt( self ) -> str:
        """Build prompt with user's available lists."""
        lists_metadata = self.storage.get_all_lists_metadata()
        return self.prompt_template.format(
            question=self.last_question_asked,
            available_lists=lists_metadata
        )

    def run_code( self ) -> dict:
        """Execute CRUD operation based on extracted intent."""
        intent = CRUDIntent.from_xml( self.prompt_response_dict.get( "raw_response" ) )

        # Dispatch to appropriate CRUD operation
        result = self.dispatcher.dispatch( intent, self.storage )

        self.code_response_dict = {
            "return_code": 0,
            "output": result
        }
        return self.code_response_dict
```

**Two-Stage Extraction** (Phi-4 primary, Claude Code fallback):

```python
def extract_intent( self, question: str, lists_metadata: list ) -> CRUDIntent:
    """Extract intent with fallback to Claude Code."""

    # Stage 1: Local Phi-4 via standard LLM factory
    factory = LlmClientFactory( debug=self.debug )
    llm = factory.get_client( self.model_name )  # From config
    response = llm.run( self.prompt )

    intent = CRUDIntent.from_xml( response )  # Uses BaseXMLModel auto-recovery

    if intent.confidence and float( intent.confidence ) >= 0.6:
        return intent

    # Stage 2: Claude Code headless fallback
    return self._extract_via_claude_code( question, lists_metadata )
```

```python
from cosa.agents.utils import voice_io

async def confirm_destructive_operation( intent: CRUDIntent, user_email: str ) -> bool:
    """Confirm before delete operations."""

    if intent.operation in ( "delete_item", "delete_list" ):
        question = f"Are you sure you want to delete from '{intent.target_list}'?"
        return await voice_io.ask_yes_no( question, default="no" )

    return True  # Non-destructive ops don't need confirmation
```
