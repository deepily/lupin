### Phase P1.2: Create API Router

**File**: `src/cosa/rest/routers/claude_code.py`

```python
"""
Claude Code Dispatcher API Router

Provides REST endpoints for dispatching Claude Code tasks from the UI.
"""

from fastapi import APIRouter, HTTPException, BackgroundTasks
from pydantic import BaseModel
from typing import Optional
from enum import Enum

from cosa.orchestration import ClaudeCodeDispatcher, Task, TaskType

router = APIRouter( prefix="/api/claude-code", tags=["claude-code"] )

# Store active dispatchers for Option B session management
active_sessions: dict[str, dict] = {}


class TaskTypeEnum( str, Enum ):
    BOUNDED = "BOUNDED"
    INTERACTIVE = "INTERACTIVE"


class DispatchRequest( BaseModel ):
    project: str
    prompt: str
    task_type: TaskTypeEnum = TaskTypeEnum.BOUNDED


class DispatchResponse( BaseModel ):
    task_id: str
    status: str
    websocket_url: str


class InjectRequest( BaseModel ):
    message: str


@router.post( "/dispatch", response_model=DispatchResponse )
async def dispatch_task( request: DispatchRequest, background_tasks: BackgroundTasks ):
    """
    Dispatch a new Claude Code task.

    Returns task_id for WebSocket subscription.
    """
    from datetime import datetime

    task_id = f"claude-code-{datetime.now().strftime('%Y%m%d-%H%M%S')}"

    task = Task(
        id=task_id,
        project=request.project,
        prompt=request.prompt,
        type=TaskType.BOUNDED if request.task_type == TaskTypeEnum.BOUNDED else TaskType.INTERACTIVE,
        max_turns=50 if request.task_type == TaskTypeEnum.BOUNDED else 200
    )

    # Create dispatcher and store for session management
    dispatcher = ClaudeCodeDispatcher()
    active_sessions[task_id] = {
        "dispatcher": dispatcher,
        "task": task,
        "status": "pending"
    }

    # Dispatch in background
    background_tasks.add_task( _run_dispatch, task_id )

    return DispatchResponse(
        task_id=task_id,
        status="dispatched",
        websocket_url=f"/ws/claude-code/{task_id}"
    )


async def _run_dispatch( task_id: str ):
    """Background task to run the dispatch."""
    session = active_sessions.get( task_id )
    if not session:
        return

    session["status"] = "running"
    dispatcher = session["dispatcher"]
    task = session["task"]

    try:
        result = await dispatcher.dispatch( task )
        session["status"] = "complete" if result.success else "failed"
        session["result"] = result
    except Exception as e:
        session["status"] = "error"
        session["error"] = str( e )


@router.post( "/{task_id}/inject" )
async def inject_message( task_id: str, request: InjectRequest ):
    """Inject a follow-up message into an Option B session."""
    session = active_sessions.get( task_id )
    if not session:
        raise HTTPException( status_code=404, detail="Session not found" )

    dispatcher = session["dispatcher"]
    success = await dispatcher.inject( task_id, request.message )

    if not success:
        raise HTTPException( status_code=400, detail="Inject failed" )

    return { "status": "injected", "task_id": task_id }


@router.post( "/{task_id}/interrupt" )
async def interrupt_session( task_id: str ):
    """Interrupt the current response in an Option B session."""
    session = active_sessions.get( task_id )
    if not session:
        raise HTTPException( status_code=404, detail="Session not found" )

    dispatcher = session["dispatcher"]
    await dispatcher.interrupt( task_id )

    return { "status": "interrupted", "task_id": task_id }


@router.post( "/{task_id}/end" )
async def end_session( task_id: str ):
    """End an Option B session gracefully."""
    session = active_sessions.get( task_id )
    if not session:
        raise HTTPException( status_code=404, detail="Session not found" )

    dispatcher = session["dispatcher"]
    await dispatcher.end_session( task_id )

    # Clean up
    del active_sessions[task_id]

    return { "status": "ended", "task_id": task_id }
```

### Phase P1.4: Wire WebSocket for Streaming

**Add to notifications.js**:

```javascript
// ========================================
// CLAUDE CODE DISPATCHER
// ========================================

setupClaudeCodeEventListeners() {
    document.getElementById( 'cc-submit' ).addEventListener( 'click', () => {
        this.submitClaudeCode();
    });

    document.getElementById( 'cc-inject-btn' ).addEventListener( 'click', () => {
        this.injectClaudeCode();
    });

    document.getElementById( 'cc-interrupt-btn' ).addEventListener( 'click', () => {
        this.interruptClaudeCode();
    });

    document.getElementById( 'cc-end-btn' ).addEventListener( 'click', () => {
        this.endClaudeCodeSession();
    });

    // Show/hide Option B controls based on selection
    document.querySelectorAll( 'input[name="cc-task-type"]' ).forEach( radio => {
        radio.addEventListener( 'change', ( e ) => {
            const optionBControls = document.getElementById( 'cc-option-b-controls' );
            optionBControls.style.display = e.target.value === 'INTERACTIVE' ? 'block' : 'none';
        });
    });
}

async submitClaudeCode() {
    const project = document.getElementById( 'cc-project' ).value;
    const prompt = document.getElementById( 'cc-prompt' ).value;
    const taskType = document.querySelector( 'input[name="cc-task-type"]:checked' ).value;

    if ( !prompt.trim() ) {
        alert( 'Please enter a task prompt' );
        return;
    }

    // Show loading
    document.getElementById( 'cc-loading' ).style.display = 'inline-block';
    document.getElementById( 'cc-submit' ).disabled = true;
    document.getElementById( 'cc-response' ).textContent = 'Dispatching...';

    try {
        const response = await fetch( '/api/claude-code/dispatch', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify( { project, prompt, task_type: taskType } )
        });

        const data = await response.json();
        this.currentClaudeCodeTaskId = data.task_id;

        // Update UI
        document.getElementById( 'cc-task-id' ).textContent = data.task_id;
        document.getElementById( 'cc-status' ).textContent = 'Running';
        document.getElementById( 'cc-session-info' ).style.display = 'flex';

        // Connect to WebSocket for streaming
        this.connectClaudeCodeWebSocket( data.task_id );

    } catch ( error ) {
        console.error( 'Claude Code dispatch failed:', error );
        document.getElementById( 'cc-response' ).textContent = `Error: ${error.message}`;
    } finally {
        document.getElementById( 'cc-loading' ).style.display = 'none';
        document.getElementById( 'cc-submit' ).disabled = false;
    }
}

connectClaudeCodeWebSocket( taskId ) {
    const wsUrl = `ws://${window.location.host}/ws/claude-code/${taskId}`;
    this.claudeCodeWs = new WebSocket( wsUrl );

    this.claudeCodeWs.onmessage = ( event ) => {
        const data = JSON.parse( event.data );
        this.handleClaudeCodeMessage( data );
    };

    this.claudeCodeWs.onclose = () => {
        document.getElementById( 'cc-status' ).textContent = 'Disconnected';
    };

    this.claudeCodeWs.onerror = ( error ) => {
        console.error( 'Claude Code WebSocket error:', error );
    };
}

handleClaudeCodeMessage( data ) {
    const responseArea = document.getElementById( 'cc-response' );

    switch ( data.type ) {
        case 'text':
            responseArea.textContent += data.content;
            break;
        case 'tool_use':
            responseArea.textContent += `\n[TOOL: ${data.name}]\n`;
            break;
        case 'tool_result':
            responseArea.textContent += `${data.content}\n`;
            break;
        case 'status':
            document.getElementById( 'cc-status' ).textContent = data.state;
            if ( data.cost_usd ) {
                document.getElementById( 'cc-cost' ).textContent = `$${data.cost_usd.toFixed(4)}`;
            }
            break;
        case 'complete':
            document.getElementById( 'cc-status' ).textContent = data.success ? 'Complete' : 'Failed';
            if ( data.cost_usd ) {
                document.getElementById( 'cc-cost' ).textContent = `$${data.cost_usd.toFixed(4)}`;
            }
            break;
        case 'error':
            responseArea.textContent += `\n[ERROR: ${data.message}]\n`;
            break;
    }

    // Auto-scroll
    responseArea.scrollTop = responseArea.scrollHeight;
}

async injectClaudeCode() {
    const message = document.getElementById( 'cc-inject-input' ).value;
    if ( !message.trim() || !this.currentClaudeCodeTaskId ) return;

    try {
        await fetch( `/api/claude-code/${this.currentClaudeCodeTaskId}/inject`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify( { message } )
        });

        document.getElementById( 'cc-inject-input' ).value = '';
        document.getElementById( 'cc-response' ).textContent += `\n[INJECTED: ${message}]\n`;

    } catch ( error ) {
        console.error( 'Inject failed:', error );
    }
}

async interruptClaudeCode() {
    if ( !this.currentClaudeCodeTaskId ) return;

    try {
        await fetch( `/api/claude-code/${this.currentClaudeCodeTaskId}/interrupt`, {
            method: 'POST'
        });

        document.getElementById( 'cc-response' ).textContent += '\n[INTERRUPTED]\n';

    } catch ( error ) {
        console.error( 'Interrupt failed:', error );
    }
}

async endClaudeCodeSession() {
    if ( !this.currentClaudeCodeTaskId ) return;

    try {
        await fetch( `/api/claude-code/${this.currentClaudeCodeTaskId}/end`, {
            method: 'POST'
        });

        document.getElementById( 'cc-status' ).textContent = 'Ended';
        this.currentClaudeCodeTaskId = null;

        if ( this.claudeCodeWs ) {
            this.claudeCodeWs.close();
        }

    } catch ( error ) {
        console.error( 'End session failed:', error );
    }
}
```
