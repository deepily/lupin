## 1. Overview

### 1.1 Problem Statement

Currently, when a user submits a question to the LUPIN system, it goes through an LLM-based router (`_get_routing_command()`) that analyzes the question and decides which agent should handle it. This works well for general use, but has limitations:

- **Latency**: Every question incurs LLM router overhead
- **Unpredictability**: Router may misclassify questions
- **No persistence**: User can't "lock in" to a specific agent for a series of related questions

### 1.2 Proposed Solution

Add a **user-level mode system** that allows users to switch into agent-specific modes. When in a mode:

- All input bypasses the LLM router
- Input routes directly to the selected agent
- Mode persists until explicitly changed
- Inputs/outputs tagged with current mode

### 1.3 Use Cases

| Use Case | Description |
|----------|-------------|
| **Math homework session** | User selects "math mode" and asks 20 math questions in a row - all go directly to MathAgent without routing overhead |
| **Calendar management** | User enters "calendar mode" to add multiple appointments without router misinterpreting requests |
| **Focused conversation** | User wants to have an extended conversation with ReceptionistAgent without interruption |

### 2.2 State Storage

**Location**: In-memory dictionary in `TodoFifoQueue`

```python
class TodoFifoQueue:
    def __init__(self, ...):
        # Existing initialization...

        # NEW: User mode state
        # Key: user_id (str)
        # Value: mode name (str) or None for system mode
        self.user_mode_state: Dict[str, Optional[str]] = {}
```

**Rationale for In-Memory Storage**:
- Consistent with existing pattern (notification I/O tracking uses in-memory dict)
- Session-only persistence (mode resets on server restart)
- No database overhead for frequent mode checks
- Simple to implement and debug

**File**: `src/cosa/rest/todo_fifo_queue.py`

**New Methods to Add:**

```python
def get_user_mode(self, user_id: str) -> Optional[str]:
    """
    Get the current mode for a user.

    Requires:
        - user_id is a non-empty string

    Ensures:
        - Returns mode string if set, None if in system mode
    """
    return self.user_mode_state.get(user_id)

def set_user_mode(self, user_id: str, mode: Optional[str]) -> str:
    """
    Set the mode for a user.

    Requires:
        - user_id is a non-empty string
        - mode is None (system) or a valid mode key

    Ensures:
        - User's mode is updated in state dictionary
        - Returns the previous mode (or None)

    Raises:
        - ValueError if mode is not a valid mode key
    """
    if mode is not None and mode not in self.MODE_TO_AGENT:
        raise ValueError(f"Invalid mode: {mode}")

    previous = self.user_mode_state.get(user_id)

    if mode is None:
        self.user_mode_state.pop(user_id, None)
    else:
        self.user_mode_state[user_id] = mode

    return previous

def clear_user_mode(self, user_id: str) -> Optional[str]:
    """
    Clear the mode for a user (return to system mode).

    Requires:
        - user_id is a non-empty string

    Ensures:
        - User is removed from mode state dictionary
        - Returns the previous mode (or None)
    """
    return self.user_mode_state.pop(user_id, None)

def get_available_modes(self) -> List[Dict[str, str]]:
    """
    Get list of available modes with display names.

    Ensures:
        - Returns list of mode dictionaries with key, display_name, description
    """
    return [
        {"key": "system", "display_name": "System", "description": "Normal LLM-based routing"},
        {"key": "math", "display_name": "Math Agent", "description": "Direct math calculations"},
        # ... etc
    ]
```

### 5.2 New Router File

**File**: `src/cosa/rest/routers/mode.py`

```python
"""
Mode management API endpoints.

Allows users to switch between agent-specific modes for direct routing.
"""

from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel
from typing import Optional, List

from cosa.rest.todo_fifo_queue import TodoFifoQueue
from cosa.rest.routers.auth import get_current_user

router = APIRouter(prefix="/api/mode", tags=["mode"])

# Request/Response models
class ModeSetRequest(BaseModel):
    mode: Optional[str] = None

class ModeResponse(BaseModel):
    user_id: str
    mode: Optional[str]
    display_name: str
    is_system_mode: bool

class ModeChangeResponse(ModeResponse):
    previous_mode: Optional[str]
    message: str

class AvailableModesResponse(BaseModel):
    modes: List[dict]


def get_todo_queue() -> TodoFifoQueue:
    """Dependency to get the singleton TodoFifoQueue instance."""
    from cosa.rest.fifo_queue_singletons import get_todo_queue as _get_queue
    return _get_queue()


@router.get("/{user_email}", response_model=ModeResponse)
async def get_mode(
    user_email: str,
    current_user = Depends(get_current_user),
    todo_queue: TodoFifoQueue = Depends(get_todo_queue)
):
    """Get current mode for a user."""
    mode = todo_queue.get_user_mode(user_email)
    display_name = _get_display_name(mode)

    return ModeResponse(
        user_id=user_email,
        mode=mode,
        display_name=display_name,
        is_system_mode=(mode is None)
    )


@router.post("/{user_email}", response_model=ModeChangeResponse)
async def set_mode(
    user_email: str,
    request: ModeSetRequest,
    current_user = Depends(get_current_user),
    todo_queue: TodoFifoQueue = Depends(get_todo_queue)
):
    """Set mode for a user."""
    try:
        previous = todo_queue.set_user_mode(user_email, request.mode)
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e))

    display_name = _get_display_name(request.mode)

    return ModeChangeResponse(
        user_id=user_email,
        mode=request.mode,
        display_name=display_name,
        is_system_mode=(request.mode is None),
        previous_mode=previous,
        message=f"Mode changed to {display_name}"
    )


@router.delete("/{user_email}", response_model=ModeChangeResponse)
async def clear_mode(
    user_email: str,
    current_user = Depends(get_current_user),
    todo_queue: TodoFifoQueue = Depends(get_todo_queue)
):
    """Clear mode (return to system mode)."""
    previous = todo_queue.clear_user_mode(user_email)

    return ModeChangeResponse(
        user_id=user_email,
        mode=None,
        display_name="System",
        is_system_mode=True,
        previous_mode=previous,
        message="Returned to System mode"
    )


@router.get("/available", response_model=AvailableModesResponse)
async def get_available_modes(
    current_user = Depends(get_current_user),
    todo_queue: TodoFifoQueue = Depends(get_todo_queue)
):
    """Get list of available modes."""
    return AvailableModesResponse(modes=todo_queue.get_available_modes())


def _get_display_name(mode: Optional[str]) -> str:
    """Get display name for a mode."""
    if mode is None:
        return "System"

    display_names = {
        "math": "Math Agent",
        "calendar": "Calendar",
        "weather": "Weather",
        "receptionist": "Receptionist",
        "todo": "Todo List",
        "datetime": "Date & Time"
    }
    return display_names.get(mode, mode.title())
```
