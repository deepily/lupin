```javascript
class HybridTTS {
  constructor(options = {}) {
    // Existing constructor code...
    this.mode = localStorage.getItem('tts-mode') || 'reliable';
    this.fallbackEnabled = true;
    this.apiUrlBatch = '/api/get-speech';
    this.apiUrlStreaming = '/api/get-speech-elevenlabs';
  }
  
  async speak(text, options = {}) {
    const mode = options.mode || this.mode;
    
    try {
      switch(mode) {
        case 'instant':
          return await this.streamingSpeak(text, options);
        case 'reliable':
          return await this.batchSpeak(text, options);
        default:
          throw new Error(`Unknown TTS mode: ${mode}`);
      }
    } catch (error) {
      if (this.fallbackEnabled) {
        return await this.fallbackSpeak(text, options, mode);
      }
      throw error;
    }
  }
}
```

```javascript
async function handleSpeechUpdate(data) {
  console.log("Received speech update:", data);
  
  if (data.text) {
    const currentMode = localStorage.getItem('tts-mode') || 'reliable';
    console.log(`Converting text to speech via ${currentMode} mode: "${data.text}"`);
    
    try {
      if (window.hybridTTS) {
        await window.hybridTTS.speak(data.text, { mode: currentMode });
      }
    } catch (error) {
      console.error('TTS failed:', error);
    }
  }
}

function updateTTSMode(newMode) {
  localStorage.setItem('tts-mode', newMode);
  console.log(`TTS mode changed to: ${newMode}`);

}
```

## Implementation Notes

### Critical Considerations
1. **Backward Compatibility**: All existing functionality must continue working
2. **Default Behavior**: System defaults to "reliable" mode for consistency
3. **Error Handling**: Fallback should be seamless to user
4. **Performance**: No significant impact on response times

### Potential Issues
1. **LocalStorage Conflicts**: Ensure no naming conflicts with existing storage
2. **Mode Mismatch**: Handle cases where saved mode is invalid
3. **API Availability**: Both endpoints must be available for testing
4. **WebSocket State**: Ensure streaming mode doesn't conflict with existing connections
