# Phase 2.6 - Cloud Run Compatibility Migration

**Date**: 2025.11.12
**Status**: Planning Phase - Blockers Documented
**Total Estimated Effort**: 30-38 hours
**Target**: Enable production Cloud Run deployment with GCP-managed services

---

## Executive Summary

Lupin currently has **3 critical blockers** preventing production Cloud Run deployment:

1. **Whisper STT GPU Dependency** - Local GPU model won't work on Cloud Run
2. **Local Database Dependencies** - Ephemeral filesystem loses all data on restart
3. **In-Memory Notification State** - Multi-instance deployment breaks response tracking

All three blockers require architectural refactoring but have clear migration paths using GCP managed services.

**Key Requirement**: All solutions must use **factory methods** to support:
- **Local development**: SQLite databases, in-memory state, local model files
- **Production (Cloud Run)**: Cloud SQL, Redis, external STT APIs

---

## Architecture Pattern: Factory Methods for Environment-Aware Services

### Design Principle

All dependency initialization must go through factory methods that detect the environment and return appropriate implementations:

```python
# src/cosa/utils/service_factory.py

def get_environment():
    """Detect if we're running locally or on Cloud Run."""
    return os.environ.get("LUPIN_ENV", "development")  # "development" | "production"

def create_database_connection():
    """Factory method for database connections."""
    env = get_environment()

    if env == "production":
        # Cloud Run: Use Cloud SQL PostgreSQL
        return CloudSQLConnection(
            connection_name=os.environ.get("CLOUD_SQL_CONNECTION"),
            database=os.environ.get("DB_NAME")
        )
    else:
        # Local development: Use SQLite
        return SQLiteConnection(
            db_path=f"{get_project_root()}/src/conf/long-term-memory/lupin-auth.db"
        )

def create_notification_state_manager():
    """Factory method for notification response tracking."""
    env = get_environment()

    if env == "production":
        # Cloud Run: Use Redis for cross-instance state
        return RedisNotificationStateManager(
            redis_url=os.environ.get("REDIS_URL")
        )
    else:
        # Local development: Use in-memory dictionary
        return InMemoryNotificationStateManager()

def create_stt_service():
    """Factory method for speech-to-text service."""
    env = get_environment()

    if env == "production":
        # Cloud Run: Use Google Cloud Speech-to-Text API
        return GoogleSTTService(
            credentials_path=os.environ.get("GOOGLE_APPLICATION_CREDENTIALS")
        )
    else:
        # Local development: Use local Whisper model
        return LocalWhisperSTTService(
            model_id="distil-whisper/distil-large-v3",
            device="cuda:0"
        )
```

### Environment Detection

**Local Development**:
- `LUPIN_ENV=development` (default if not set)
- Uses SQLite, in-memory state, local GPU models
- No GCP credentials needed

**Cloud Run Production**:
- `LUPIN_ENV=production` (set via `gcloud run deploy --set-env-vars`)
- Uses Cloud SQL, Redis, external APIs
- GCP credentials via service account

```python
from abc import ABC, abstractmethod
from typing import Optional

class STTService(ABC):
    """Abstract base class for speech-to-text services."""

    @abstractmethod
    async def transcribe(self, audio_bytes: bytes, language: str = "en") -> str:
        """Transcribe audio bytes to text."""
        pass

class LocalWhisperSTTService(STTService):
    """Local Whisper model for development."""

    def __init__(self, model_id: str, device: str = "cuda:0"):
        import torch
        from transformers import pipeline

        torch_dtype = torch.float16 if torch.cuda.is_available() else torch.float32
        self.pipe = pipeline(
            "automatic-speech-recognition",
            model=model_id,
            torch_dtype=torch_dtype,
            device=device
        )

    async def transcribe(self, audio_bytes: bytes, language: str = "en") -> str:
        """Transcribe using local Whisper model."""
        # Existing implementation
        result = self.pipe(audio_bytes)
        return result["text"]

class GoogleSTTService(STTService):
    """Google Cloud Speech-to-Text API for production."""

    def __init__(self, credentials_path: Optional[str] = None):
        from google.cloud import speech_v1

        if credentials_path:
            os.environ["GOOGLE_APPLICATION_CREDENTIALS"] = credentials_path

        self.client = speech_v1.SpeechClient()

    async def transcribe(self, audio_bytes: bytes, language: str = "en") -> str:
        """Transcribe using Google Cloud Speech-to-Text."""
        from google.cloud import speech_v1

        audio = speech_v1.RecognitionAudio(content=audio_bytes)
        config = speech_v1.RecognitionConfig(
            encoding=speech_v1.RecognitionConfig.AudioEncoding.LINEAR16,
            sample_rate_hertz=16000,
            language_code=language,
            model="latest_long",  # Best quality model
        )

        response = self.client.recognize(config=config, audio=audio)

        # Return first transcript
        if response.results:
            return response.results[0].alternatives[0].transcript
        return ""
```

```python
from abc import ABC, abstractmethod
from typing import Optional, Dict, Any, List

class DatabaseConnection(ABC):
    """Abstract base class for database connections."""

    @abstractmethod
    async def execute_query(self, query: str, params: Optional[Dict] = None) -> List[Dict]:
        """Execute SQL query and return results."""
        pass

    @abstractmethod
    async def execute_update(self, query: str, params: Optional[Dict] = None) -> int:
        """Execute SQL update/insert/delete and return affected rows."""
        pass

    @abstractmethod
    async def close(self):
        """Close database connection."""
        pass

class SQLiteConnection(DatabaseConnection):
    """SQLite connection for local development."""

    def __init__(self, db_path: str):
        import aiosqlite
        self.db_path = db_path
        self._connection = None

    async def connect(self):
        """Open SQLite connection."""
        import aiosqlite
        self._connection = await aiosqlite.connect(self.db_path)
        self._connection.row_factory = aiosqlite.Row

    async def execute_query(self, query: str, params: Optional[Dict] = None) -> List[Dict]:
        """Execute query and return results as list of dicts."""
        if not self._connection:
            await self.connect()

        async with self._connection.execute(query, params or {}) as cursor:
            rows = await cursor.fetchall()
            return [dict(row) for row in rows]

    async def execute_update(self, query: str, params: Optional[Dict] = None) -> int:
        """Execute update and return affected rows."""
        if not self._connection:
            await self.connect()

        async with self._connection.execute(query, params or {}) as cursor:
            await self._connection.commit()
            return cursor.rowcount

    async def close(self):
        """Close SQLite connection."""
        if self._connection:
            await self._connection.close()

class CloudSQLConnection(DatabaseConnection):
    """Cloud SQL PostgreSQL connection for production."""

    def __init__(self, connection_name: str, database: str, user: str, password: str):
        from google.cloud.sql.connector import Connector
        import asyncpg

        self.connection_name = connection_name  # "project:region:instance"
        self.database = database
        self.user = user
        self.password = password
        self._connector = Connector()
        self._pool = None

    async def connect(self):
        """Create connection pool to Cloud SQL."""
        import asyncpg

        async def getconn():
            """Create connection via Cloud SQL Connector."""
            conn = await self._connector.connect_async(
                self.connection_name,
                "asyncpg",
                user=self.user,
                password=self.password,
                db=self.database
            )
            return conn

        # Create connection pool
        self._pool = await asyncpg.create_pool(
            min_size=1,
            max_size=5,
            command_timeout=60,
            setup=getconn
        )

    async def execute_query(self, query: str, params: Optional[Dict] = None) -> List[Dict]:
        """Execute query and return results as list of dicts."""
        if not self._pool:
            await self.connect()

        async with self._pool.acquire() as conn:
            rows = await conn.fetch(query, *(params.values() if params else []))
            return [dict(row) for row in rows]

    async def execute_update(self, query: str, params: Optional[Dict] = None) -> int:
        """Execute update and return affected rows."""
        if not self._pool:
            await self.connect()

        async with self._pool.acquire() as conn:
            result = await conn.execute(query, *(params.values() if params else []))
            # Parse result like "UPDATE 3" → 3
            return int(result.split()[-1]) if result else 0

    async def close(self):
        """Close connection pool."""
        if self._pool:
            await self._pool.close()
        await self._connector.close_async()
```

```python
# When notification is sent (POST /api/notify?response_required=true):
response_event = asyncio.Event()
pending_responses[notification_id] = {
    "event"         : response_event,
    "response_data" : None
}

# SSE stream blocks waiting for response:
await response_event.wait()  # ❌ Will wait forever if response goes to different instance!

# When response is received (POST /api/notification-response):
if notification_id in pending_responses:
    pending_responses[notification_id]["response_data"] = response_value
    pending_responses[notification_id]["event"].set()  # Wake up SSE stream!
else:
    # ❌ Response lost! Not found because it's on a different instance
    pass
```

**Usage**: SSE (Server-Sent Events) blocking flow for response-required notifications

### Cloud Run Blocker Analysis

| Issue | Impact |
|-------|--------|
| **Multi-instance deployment** | Each Cloud Run instance has its own `pending_responses` dict |
| **Request routing** | Request A (send notification) and Request B (receive response) may hit different instances |
| **No shared memory** | Instances cannot share Python dictionaries |
| **Lost responses** | If instance handling SSE stream dies, response is lost |
| **Race conditions** | Instance scaling during notification flow causes failures |

**Example Failure Scenario**:
```
Timeline of Multi-Instance Failure:

1. Claude Code sends notification with response_required=true
   → Request routed to Instance A
   → Instance A creates pending_responses[123] = {event: Event(), response_data: None}
   → Instance A starts SSE stream, blocks waiting for response

2. User responds in browser (clicks "Approve" button)
   → Request routed to Instance B (load balancer round-robin)
   → Instance B receives POST /api/notification-response with notification_id=123
   → Instance B looks for pending_responses[123]
   → NOT FOUND! (It only exists on Instance A's memory)
   → Response is discarded (logged as error)

3. Instance A still waiting...
   → SSE stream times out after 5 minutes
   → Claude Code receives timeout error instead of user's response
   → User action was lost

Result: Broken notification flow, poor UX, lost responses
```

### Solution: Redis Pub/Sub with Factory Method

**Recommended**: Cloud Memorystore for Redis (managed Redis service)

**Pros**:
- ✅ Multi-instance compatible (shared state across all instances)
- ✅ Low latency (1-5ms for pub/sub messages)
- ✅ Built-in pub/sub for SSE wakeup across instances
- ✅ Atomic operations (race condition safe)
- ✅ Auto-expiry (TTL) for automatic cleanup
- ✅ Fully managed by GCP (no maintenance)
- ✅ High availability (automatic failover)

**Cons**:

- Cost (~$12-25/month for 1GB basic tier)

```python
from abc import ABC, abstractmethod
from typing import Optional, Any
import asyncio

class NotificationStateManager(ABC):
    """Abstract base class for notification response state management."""

    @abstractmethod
    async def create_pending_response(self, notification_id: str, timeout_seconds: int):
        """Mark notification as waiting for response."""
        pass

    @abstractmethod
    async def set_response(self, notification_id: str, response_value: Any):
        """Store response value and wake up any waiting SSE streams."""
        pass

    @abstractmethod
    async def wait_for_response(self, notification_id: str, timeout_seconds: int) -> Optional[Any]:
        """Wait for response (blocks until response received or timeout)."""
        pass

    @abstractmethod
    async def cleanup_notification(self, notification_id: str):
        """Remove notification from pending state."""
        pass

class InMemoryNotificationStateManager(NotificationStateManager):
    """In-memory state manager for local development (single instance only)."""

    def __init__(self):
        # {notification_id: {"event": asyncio.Event(), "response_data": None}}
        self._pending_responses = {}

    async def create_pending_response(self, notification_id: str, timeout_seconds: int):
        """Create pending response entry with asyncio.Event."""
        response_event = asyncio.Event()
        self._pending_responses[notification_id] = {
            "event": response_event,
            "response_data": None
        }

    async def set_response(self, notification_id: str, response_value: Any):
        """Set response value and wake up SSE stream."""
        if notification_id in self._pending_responses:
            self._pending_responses[notification_id]["response_data"] = response_value
            self._pending_responses[notification_id]["event"].set()
        else:
            raise ValueError(f"Notification {notification_id} not found in pending responses")

    async def wait_for_response(self, notification_id: str, timeout_seconds: int) -> Optional[Any]:
        """Wait for response with timeout."""
        if notification_id not in self._pending_responses:
            raise ValueError(f"Notification {notification_id} not found")

        try:
            await asyncio.wait_for(
                self._pending_responses[notification_id]["event"].wait(),
                timeout=timeout_seconds
            )
            return self._pending_responses[notification_id]["response_data"]
        except asyncio.TimeoutError:
            return None  # Timeout - use default response

    async def cleanup_notification(self, notification_id: str):
        """Remove notification from pending state."""
        if notification_id in self._pending_responses:
            del self._pending_responses[notification_id]

class RedisNotificationStateManager(NotificationStateManager):
    """Redis-based state manager for production (multi-instance compatible)."""

    def __init__(self, redis_url: str):
        import redis.asyncio as redis

        self.redis = redis.from_url(redis_url, decode_responses=True)
        self.pubsub = None  # Lazy-initialized per connection

    async def create_pending_response(self, notification_id: str, timeout_seconds: int):
        """Store notification state in Redis with TTL."""
        import json

        await self.redis.setex(
            f"notification:{notification_id}",
            timeout_seconds,
            json.dumps({"status": "pending", "response_data": None})
        )

    async def set_response(self, notification_id: str, response_value: Any):
        """Store response and publish event to wake up SSE streams on ANY instance."""
        import json

        # Update state in Redis
        key = f"notification:{notification_id}"
        value = json.dumps({"status": "responded", "response_data": response_value})

        # Check if notification exists
        exists = await self.redis.exists(key)
        if not exists:
            raise ValueError(f"Notification {notification_id} not found or expired")

        await self.redis.set(key, value)

        # Publish event to wake up SSE stream (on ANY instance!)
        channel = f"notification_response:{notification_id}"
        await self.redis.publish(channel, json.dumps(response_value))

    async def wait_for_response(self, notification_id: str, timeout_seconds: int) -> Optional[Any]:
        """
        Wait for response using Redis pub/sub.

        This works across multiple Cloud Run instances:
        1. Instance A creates pending response and subscribes to Redis channel
        2. User responds, request goes to Instance B
        3. Instance B publishes response to Redis channel
        4. Instance A receives pub/sub message and wakes up SSE stream
        5. Response delivered successfully!
        """
        import json
        import asyncio

        # Subscribe to notification response channel
        channel = f"notification_response:{notification_id}"
        pubsub = self.redis.pubsub()
        await pubsub.subscribe(channel)

        try:
            # Wait for message with timeout
            async def listen_for_response():
                async for message in pubsub.listen():
                    if message["type"] == "message":
                        return json.loads(message["data"])

            response_data = await asyncio.wait_for(
                listen_for_response(),
                timeout=timeout_seconds
            )
            return response_data

        except asyncio.TimeoutError:
            # Timeout - check if response was set while we were timing out
            notification_data = await self.redis.get(f"notification:{notification_id}")
            if notification_data:
                data = json.loads(notification_data)
                if data["status"] == "responded":
                    return data["response_data"]

            return None  # True timeout - use default

        finally:
            await pubsub.unsubscribe(channel)
            await pubsub.close()

    async def cleanup_notification(self, notification_id: str):
        """Remove notification from Redis."""
        await self.redis.delete(f"notification:{notification_id}")

    async def close(self):
        """Close Redis connection."""
        await self.redis.close()
```
